elf_destiny_utility.01 = { # re-enable toast notifications
	hidden = yes

	immediate = {
		remove_global_variable = disable_toast_notifications
	}
}

scripted_effect toggle_wandering_setting = {
	if = {
		limit = {global_var:$WANDERING_SETTING$ = yes}
		set_global_variable = {
			name = $WANDERING_SETTING$ 
			value = no
		}
	}
	else = {
		set_global_variable = {
			name = $WANDERING_SETTING$ 
			value = yes
		}
	}
}
